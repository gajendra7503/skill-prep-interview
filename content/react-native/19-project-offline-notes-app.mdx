Most real mobile apps share the same skeleton: a list screen, a detail/edit
screen, local persistence so data survives app restarts, and navigation
between them. Let's build that skeleton as a notes app.

## 1. Navigation structure

```jsx
// App.js
import { NavigationContainer } from "@react-navigation/native";
import { createNativeStackNavigator } from "@react-navigation/native-stack";
import NotesListScreen from "./screens/NotesListScreen";
import NoteEditorScreen from "./screens/NoteEditorScreen";

const Stack = createNativeStackNavigator();

export default function App() {
  return (
    <NavigationContainer>
      <Stack.Navigator>
        <Stack.Screen name="NotesList" component={NotesListScreen} options={{ title: "Notes" }} />
        <Stack.Screen name="NoteEditor" component={NoteEditorScreen} options={{ title: "Edit Note" }} />
      </Stack.Navigator>
    </NavigationContainer>
  );
}
```

## 2. A persistence hook over AsyncStorage

```jsx
import { useCallback, useEffect, useState } from "react";
import AsyncStorage from "@react-native-async-storage/async-storage";

const STORAGE_KEY = "notes";

export function useNotes() {
  const [notes, setNotes] = useState([]);
  const [loaded, setLoaded] = useState(false);

  useEffect(() => {
    AsyncStorage.getItem(STORAGE_KEY).then((raw) => {
      setNotes(raw ? JSON.parse(raw) : []);
      setLoaded(true);
    });
  }, []);

  const persist = useCallback((next) => {
    setNotes(next);
    AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(next));
  }, []);

  const addNote = useCallback(
    (text) => {
      persist([{ id: Date.now().toString(), text, updatedAt: Date.now() }, ...notes]);
    },
    [notes, persist]
  );

  const updateNote = useCallback(
    (id, text) => {
      persist(notes.map((n) => (n.id === id ? { ...n, text, updatedAt: Date.now() } : n)));
    },
    [notes, persist]
  );

  const deleteNote = useCallback(
    (id) => {
      persist(notes.filter((n) => n.id !== id));
    },
    [notes, persist]
  );

  return { notes, loaded, addNote, updateNote, deleteNote };
}
```

This mirrors the `useLocalStorage` hook from the React course, swapping the
synchronous `localStorage` API for AsyncStorage's asynchronous one — the
pattern (load once on mount, persist on every write) is identical.

## 3. The list screen, with FlatList performance in mind

```jsx
function NotesListScreen({ navigation }) {
  const { notes, loaded, deleteNote } = useNotes();

  if (!loaded) return <ActivityIndicator style={{ flex: 1 }} />;

  return (
    <FlatList
      data={notes}
      keyExtractor={(item) => item.id}
      renderItem={({ item }) => (
        <Pressable
          onPress={() => navigation.navigate("NoteEditor", { id: item.id })}
          style={styles.row}
        >
          <Text numberOfLines={1} style={styles.rowText}>
            {item.text || "Untitled note"}
          </Text>
          <Pressable onPress={() => deleteNote(item.id)} hitSlop={8}>
            <Text style={styles.deleteText}>Delete</Text>
          </Pressable>
        </Pressable>
      )}
      ListEmptyComponent={<Text style={styles.empty}>No notes yet — add one.</Text>}
    />
  );
}
```

`keyExtractor` using the note's stable `id` (not array index) is exactly the
"stable key" principle from the React fundamentals course, and matters even
more here — `FlatList` uses it to recycle row components as you scroll,
so an unstable key causes visible flicker and wasted re-renders.

### What this project demonstrates

- **Navigation + screen params** — `NoteEditorScreen` reads `route.params.id`
  to know which note it's editing, the same pattern used across almost every
  stack-based mobile app.
- **Async persistence with a loading flag** — `loaded` prevents rendering an
  empty list for a split second before AsyncStorage resolves, avoiding a
  jarring flash of "no notes" on every app launch.
- **FlatList over a plain `.map()` in a ScrollView** — FlatList only renders
  what's on (or near) screen, which is what keeps a notes app with hundreds
  of entries scrolling smoothly.

### Extend it yourself

- Add a search bar that filters `notes` by text before passing them to
  `FlatList`'s `data` prop.
- Replace AsyncStorage with a SQLite-backed store (from the offline-first
  lesson) and compare how much of the hook's public API has to change —
  ideally none of it, since callers only see `notes`/`addNote`/`updateNote`/`deleteNote`.
- Add pull-to-refresh (`refreshControl`) that re-reads from storage, useful
  once this app syncs with a real backend.
